"""LumaRetouch semantic-only documentation decorators.

These intentionally preserve existing static docstrings and perform no parsing,
generation, generic-model introspection, tokenization, or regex substitution.
Only the two decorator call signatures used by pinned OneFormer/Swin are part of
this private fork's API. Other Transformers documentation APIs are unsupported.
"""


def auto_docstring(obj=None, *, custom_intro=None, custom_args=None, checkpoint=None):
    return (lambda value: value) if obj is None else obj


def auto_class_docstring(cls, custom_intro=None, custom_args=None, checkpoint=None):
    return cls
